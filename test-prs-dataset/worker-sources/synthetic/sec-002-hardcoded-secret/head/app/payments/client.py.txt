import httpx

PAYMENT_API_KEY = "sk_live_51Hc8x2F9sA0bQnR7pLmZq"

def charge(amount_cents, customer_id):
    response = httpx.post(
        "https://payments.example.com/v1/charges",
        json={"amount": amount_cents, "customer": customer_id},
        headers={"Authorization": f"Bearer {PAYMENT_API_KEY}"},
    )
    response.raise_for_status()
    return response.json()
