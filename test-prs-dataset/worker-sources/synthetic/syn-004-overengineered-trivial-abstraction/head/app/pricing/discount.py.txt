class DiscountStrategy:
    def apply(self, amount):
        raise NotImplementedError

class TenPercentDiscountStrategy(DiscountStrategy):
    def apply(self, amount):
        return amount * 0.9

class DiscountCalculator:
    def __init__(self, strategy: DiscountStrategy):
        self._strategy = strategy

    def calculate(self, amount):
        return self._strategy.apply(amount)

def apply_order_discount(amount):
    calculator = DiscountCalculator(TenPercentDiscountStrategy())
    return calculator.calculate(amount)
